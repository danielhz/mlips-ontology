The number of message-passing layers in MACE is a purely
architectural hyperparameter.
\begin{lstlisting}[language=turtle]
ex:hp-num-layers a mlips:Hyperparameter,
        mlips:ArchitecturalHyperparameter ;
    mlips:hyperparameterName "num_layers" ;
    mlips:hyperparameterDatatype xsd:integer ;
    mlips:defaultValue "2"^^xsd:integer .
\end{lstlisting}
