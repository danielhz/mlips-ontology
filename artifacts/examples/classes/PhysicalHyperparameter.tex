The cutoff radius is a physical hyperparameter: its value bounds the
physical interaction range of the potential.
\begin{lstlisting}[language=turtle]
ex:hp-cutoff-radius a mlips:Hyperparameter,
        mlips:PhysicalHyperparameter ;
    mlips:hyperparameterName "R_cut" ;
    mlips:hyperparameterDatatype xsd:double ;
    mlips:defaultValue "5.0"^^xsd:double .
\end{lstlisting}
