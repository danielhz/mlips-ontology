The optimiser learning rate is a training hyperparameter — it
governs how the model is fitted, not what the model is.
\begin{lstlisting}[language=turtle]
ex:hp-learning-rate a mlips:Hyperparameter,
        mlips:TrainingHyperparameter ;
    mlips:hyperparameterName "learning_rate" ;
    mlips:hyperparameterDatatype xsd:double ;
    mlips:defaultValue "0.001"^^xsd:double .
\end{lstlisting}
