% Manually-laid-out ontology overview.
% Module membership is encoded by node fill colour (no cluster boxes):
%   * Method module        → blue   (\algofill)
%   * Training Data module → green  (\trainfill)
%   * Benchmark module     → orange (\benchfill)
% Cross-module relationships are dashed.
%
% Layout invariants:
%   * Six rows defined by the central column (Implementation, MLIPMethod,
%     MLIPRun, TrainedModel, MaterialSystem, TrainingDataset).  All
%     central-column nodes share x = 5, the same minimum width, and a
%     uniform vertical spacing of 1.3 cm.
%   * Left column (Library, SimulationType, AtomicConfiguration) shares
%     west x = 0 (anchor=west) so the gap between them and the central
%     column scales with each node's natural width — wider labels stay
%     close to the central column, narrower ones leave room for the
%     incoming edge label.
%   * Right column (HyperparameterSetting, BenchmarkStudy,
%     BenchmarkResult, AccuracyMetric) shares east x = 16.5.
%   * Nodes that share a row share the same y.
\begin{tikzpicture}[
    x=1cm, y=1cm,
    every node/.style={font=\footnotesize},
    % text height + text depth force every node label to claim the same
    % vertical metrics regardless of whether the word has descenders ("g")
    % or only x-height letters ("c"), so all baselines line up.
    methodnode/.style={draw, rounded corners=2pt, fill=algofill,
        minimum height=0.55cm, inner sep=3pt, align=center,
        font=\footnotesize, text height=1.6ex, text depth=0.4ex},
    trainnode/.style={draw, rounded corners=2pt, fill=trainfill,
        minimum height=0.55cm, inner sep=3pt, align=center,
        font=\footnotesize, text height=1.6ex, text depth=0.4ex},
    benchnode/.style={draw, rounded corners=2pt, fill=benchfill,
        minimum height=0.55cm, inner sep=3pt, align=center,
        font=\footnotesize, text height=1.6ex, text depth=0.4ex},
    % Forces a fixed width on the central-column nodes.
    centerwidth/.style={minimum width=2.5cm},
    edge/.style={->, >=stealth, thin},
    crossedge/.style={->, >=stealth, thin, dashed},
    edgelabel/.style={font=\tiny, fill=white, inner sep=0.5pt},
]

% Color definitions (copy to preamble if not already defined):
% \definecolor{algofill}{HTML}{A6CEE3}
% \definecolor{trainfill}{HTML}{B2DF8A}
% \definecolor{benchfill}{HTML}{FDBF6F}

  % ─── Central column (defines the rows; six nodes at x = 5) ───────
  % Uniform vertical spacing: 1.3 cm.  Same minimum width.
  \node[methodnode, centerwidth] (Implementation)  at (5, 7.0) {Implementation};
  \node[methodnode, centerwidth] (MLIPMethod)      at (5, 5.7) {MLIPMethod};
  \node[methodnode, centerwidth] (MLIPRun)         at (5, 4.4) {MLIPRun};
  \node[methodnode, centerwidth] (TrainedModel)    at (5, 3.1) {TrainedModel};
  \node[trainnode,  centerwidth] (MaterialSystem)  at (5, 1.8) {MaterialSystem};
  \node[trainnode,  centerwidth] (TrainingDataset) at (5, 0.5) {TrainingDataset};

  % ─── Left column (west edges aligned at x = -0.1) ────────────────
  \node[methodnode, anchor=west] (Library)              at (-0.1, 7.0) {Library};
  \node[methodnode, anchor=west] (SimulationType)       at (-0.1, 5.7) {SimulationType};
  \node[trainnode,  anchor=west] (AtomicConfiguration)  at (-0.1, 1.8) {AtomicConfiguration};

  % ─── Right column (east edges aligned at x = 16.5; the three
  %     orange Benchmark nodes share the same minimum width). ──────
  \node[methodnode, anchor=east] (HyperparameterSetting) at (16.5, 4.4) {HyperparameterSetting};
  \node[benchnode,  anchor=east, minimum width=2.5cm] (BenchmarkStudy)
    at (16.5, 3.1) {BenchmarkStudy};
  \node[benchnode,  anchor=east, minimum width=2.5cm] (BenchmarkResult)
    at (16.5, 1.8) {BenchmarkResult};
  \node[benchnode,  anchor=east, minimum width=2.5cm] (AccuracyMetric)
    at (16.5, 0.5) {AccuracyMetric};

  % ─── Method-module nodes right of the central column.  Same minimum
  %     width so their west edges line up. ────────────────────────────
  \node[methodnode, minimum width=2.5cm] (FunctionalForm) at (9.75, 7.0) {FunctionalForm};
  \node[methodnode, minimum width=2.5cm] (LossFunction)   at (9.75, 5.7) {LossFunction};
  \node[methodnode, minimum width=2.5cm] (Hyperparameter) at (9.75, 4.4) {Hyperparameter};

  % ─── Training-data-module nodes in the bottom row (y = 0.5) ──────
  \node[trainnode] (DFTCalculation)  at (9.25,  0.5) {DFTCalculation};
  \node[trainnode] (DFTSettings)     at (12.75, 0.5) {DFTSettings};

  % ─── Method-internal edges.  Incoming arrows on FunctionalForm,
  %     LossFunction, Hyperparameter, and TrainedModel are anchored to
  %     the middle of the side they approach. ─────────────────────────
  \draw[edge] (MLIPMethod) -- node[edgelabel] {hasFunctionalForm} (FunctionalForm.west);
  \draw[edge] (MLIPMethod) -- node[edgelabel] {hasLossFunction} (LossFunction.west);
  \draw[edge] (MLIPMethod) -- node[edgelabel] {hasHyperparameter} (Hyperparameter.west);
  \draw[edge] (MLIPMethod) -- node[edgelabel] {hasImplementation} (Implementation);
  \draw[edge] (MLIPMethod) -- node[edgelabel] {supportsSimulation} (SimulationType);
  \draw[edge] (Implementation) -- node[edgelabel] {implementedIn} (Library);
  \draw[edge] (HyperparameterSetting) -- node[edgelabel] {forHyperparameter} (Hyperparameter.east);
  \draw[edge] (MLIPRun) -- node[edgelabel] {appliesMethod} (MLIPMethod);
  \draw[edge] (MLIPRun) -- node[edgelabel] {produces} (TrainedModel.north);

  % ─── Training-Data-internal edges ────────────────────────────────
  \draw[edge] (TrainingDataset) -- node[edgelabel] {hasDFTCalculation} (DFTCalculation);
  \draw[edge] (TrainingDataset) -- node[edgelabel] {coversMaterial} (MaterialSystem);
  % hasConfiguration: leaves TrainingDataset at 2/3 down its west side
  % (so it doesn't collide with runsOn which arrives at 1/3 down the
  % same side), heading west (out=180), and enters AtomicConfiguration
  % from the south (in=270, ending pointing N).
  \draw[edge] ($(TrainingDataset.north west)!2/3!(TrainingDataset.south west)$)
    to[out=180, in=270]
    node[edgelabel, pos=0.6] {hasConfiguration} (AtomicConfiguration.south);
  \draw[edge] (DFTCalculation) -- node[edgelabel] {hasDFTSettings} (DFTSettings);

  % ─── Benchmark-internal edges ────────────────────────────────────
  \draw[edge] (BenchmarkStudy) -- node[edgelabel] {hasResult} (BenchmarkResult);
  \draw[edge] (BenchmarkResult) -- node[edgelabel] {hasAccuracyMetric} (AccuracyMetric);

  % ─── Cross-module edges (dashed) ─────────────────────────────────
  % runsOn: routed as a U-shape with two horizontal end segments and a
  % vertical middle segment in the corridor between AtomicConfiguration
  % and the central column.  The horizontal end segments make the
  % arrow tip orthogonal to TrainingDataset's west side (and the start
  % orthogonal to MLIPRun's west side too, for symmetry).
  \coordinate (runsOnEnd)
    at ($(TrainingDataset.north west)!1/3!(TrainingDataset.south west)$);
  \coordinate (runsOnTurn) at ($(runsOnEnd) + (-0.75, 0)$);
  \draw[crossedge, rounded corners=4pt]
    (MLIPRun.west) -- (3, 4.4) --
    node[edgelabel] {runsOn}
    (runsOnTurn) -- (runsOnEnd);
  \draw[crossedge, bend right=15] (BenchmarkResult) to
    node[edgelabel, pos=0.5] {evaluatesModel} (TrainedModel.east);
  \draw[crossedge] (BenchmarkResult) -- node[edgelabel] {targetMaterial} (MaterialSystem);

\end{tikzpicture}
